# Extra apt packages for the Jetson Orin robot PC, on top of ../apt-packages.txt.
# Installed by provision.sh phase "jetson".

# ur_rtde and moderngl have no arm64 wheels on PyPI and build from source inside the venv
libboost-all-dev
cmake
ninja-build
pkg-config

# NVIDIA's torch wheels link against OpenBLAS; python3-dev for the sdist builds above
libopenblas-dev
python3-dev

# The JetPack components. The Jetson ISO flash installs the BSP only (no CUDA, cuDNN or TensorRT) and Isaac ROS 4.6
# needs them; NVIDIA's guide: apt update, dist-upgrade, reboot, then this metapackage (about an hour). v1 left it out
# to save space on the 64 GB eMMC; root is on the 512 GB NVMe now.
nvidia-jetpack

# fdtget/fdtput: the ttyTHS1 PIO device tree for the RoboClaw link (provision.sh uart1_pio_dtb)
device-tree-compiler

# jetson-stats (jtop) is installed with pip in the jetson phase, which runs before base: the image has no pip
python3-pip
